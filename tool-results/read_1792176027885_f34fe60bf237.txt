   188→/* ============ Mappers: bungkus data const (as const) ke interface ketat ============ */
   189→
   190→function toNavItems(links: readonly { label: string; href: string; xlOnly?: boolean }[]): NavItem[] {
   191→  return links.map((l) => {
   192→    const item: NavItem = { label: l.label, href: l.href };
   193→    if (l.xlOnly) item.xlOnly = true;
   194→    return item;
   195→  });
   196→}
   197→
   198→function toStats(stats: readonly { value: number; suffix: string; label: string }[]): StatItem[] {
   199→  return stats.map((s) => ({ value: s.value, suffix: s.suffix, label: s.label }));
   200→}
   201→
   202→function toFloatCards(
   203→  cards: readonly { title: string; time: string; icon: string; accent: string }[]
   204→): FloatCard[] {
   205→  return cards.map((c) => ({ title: c.title, time: c.time, icon: c.icon, accent: c.accent }));
   206→}
   207→
   208→function toMarquee(items: readonly string[]): string[] {
   209→  return items.map((item) => item);
   210→}
   211→
   212→function toWhyItems(
   213→  rows: readonly { title: string; description: string; icon: string }[]
   214→): WhyItem[] {
   215→  return rows.map((r) => ({
   216→    title: r.title,
   217→    description: r.description,
   218→    icon: r.icon,
   219→  }));
   220→}
   221→
   222→function toProcessSteps(
   223→  rows: readonly { step: string; title: string; description: string; icon: string }[]
   224→): ProcessStepItem[] {
   225→  return rows.map((r) => ({
   226→    step: r.step,
   227→    title: r.title,
   228→    description: r.description,
   229→    icon: r.icon,
   230→  }));
   231→}
   232→
   233→function toFaqs(
   234→  rows: readonly { question: string; answer: string }[]
   235→): FaqItem[] {
   236→  return rows.map((r) => ({ question: r.question, answer: r.answer }));
   237→}
   238→
   239→function toTestimonials(
   240→  rows: readonly {
   241→    name: string;
   242→    role: string;
   243→    content: string;
   244→    rating: number;
   245→    service: string;
   246→  }[]
   247→): TestimonialItem[] {
   248→  return rows.map((r) => ({
   249→    name: r.name,
   250→    role: r.role,
   251→    content: r.content,
   252→    rating: r.rating,
   253→    service: r.service,
   254→  }));
   255→}
   256→
   257→function toInternational(
   258→  rows: readonly { country: string; service: string }[]
   259→): IntlItem[] {
   260→  return rows.map((r) => ({ country: r.country, service: r.service }));
   261→}
   262→
   263→function toComparison(c: {
   264→  us: { name: string; points: readonly string[] };
   265→  others: { name: string; points: readonly string[] };
   266→}): ComparisonPair {
   267→  return {
   268→    us: { name: c.us.name, points: [...c.us.points] },
   269→    others: { name: c.others.name, points: [...c.others.points] },
   270→  };
   271→}
   272→
   273→function toPhaseTexts(rows: readonly PhaseMeta[]): PhaseText[] {
   274→  return rows.map((p) => ({
   275→    id: p.id,
   276→    name: p.name,
   277→    tagline: p.tagline,
   278→    description: p.description,
   279→  }));
   280→}
